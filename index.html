// ✅ Automatically send code to GitHub function sendToGitHub(code) { const token = document.getElementById('ghToken').value; if (!token) { alert("Please enter your GitHub Personal Access Token!"); return; } const repo = "MyARGame"; const username = "LuckEly8"; const filePath = "app/src/main/java/com/yourcompany/sphereapp3/MainActivity.kt"; // 1. First, GET the sha of the existing file fetch(`https://api.github.com/repos/${username}/${repo}/contents/${filePath}`, { method: "GET", headers: { "Authorization": `Bearer ${token}`, "Accept": "application/vnd.github.v3+json" } }) .then(response => response.json()) .then(fileData => { const sha = fileData.sha; // Find the sha // 2. Now, UPDATE the file with the sha return fetch(`https://api.github.com/repos/${username}/${repo}/contents/${filePath}`, { method: "PUT", headers: { "Authorization": `Bearer ${token}`, "Accept": "application/vnd.github.v3+json" }, body: JSON.stringify({ message: "Generated by CodeRunner Agent", content: btoa(unescape(encodeURIComponent(code))), sha: sha }) }); }) .then(response => response.json()) .then(data => { if (data.content) { alert("Code sent to GitHub! APK will be built in 5-10 minutes."); setTimeout(() => { const apkLink = `https://github.com/${username}/${repo}/actions`; window.open(apkLink, "_blank"); }, 3000); } else { alert("Error sending to GitHub: " + JSON.stringify(data)); } }) .catch(error => { alert("Error: " + error); }); }
